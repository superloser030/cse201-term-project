#include <cstdio>
#include <iostream>
#include <utility>

#ifdef _WIN32
#include <io.h>
namespace { bool stdinIsTerminal() { return _isatty(_fileno(stdin)) != 0; } }
#else
#include <unistd.h>
namespace { bool stdinIsTerminal() { return isatty(fileno(stdin)) != 0; } }
#endif

ConsoleIO::ConsoleIO(std::function<void()> snapshotHandler)
    : snapshotHandler_(std::move(snapshotHandler)), echo_(!stdinIsTerminal()) {}

std::string ConsoleIO::nextLine(bool allowCancel) {
    while (true) {
        std::string line;
        if (!std::getline(std::cin, line)) throw InputClosedException();
        const std::string value = text::trim(line);
        if (echo_) std::cout << value << '\n';           // 파일 입력일 때 실행 기록에 입력값 표시

        if (value == "/") {                              // REQ10.1: 어느 입력에서든 snapshot
            if (snapshotHandler_) snapshotHandler_();
            std::cout << "(continue input) ";
            continue;
        }
        if (allowCancel && text::toLower(value) == CANCEL_TOKEN) throw CancelException();
        return value;
    }
}

long long ConsoleIO::parseNumber(const std::string& s, long long min, long long max) {
    if (s.empty()) throw InvalidInputException(MsgId::ERR_INVALID_INPUT);
    long long value = 0;
    try {
        std::size_t pos = 0;
        value = std::stoll(s, &pos);
        if (pos != s.size()) throw InvalidInputException(MsgId::ERR_INVALID_INPUT);
    } catch (const InvalidInputException&) {
        throw;
    } catch (const std::exception&) {                    // invalid_argument, out_of_range
        throw InvalidInputException(MsgId::ERR_INVALID_INPUT);
    }
    if (value < min || value > max) throw InvalidInputException(MsgId::ERR_INVALID_INPUT);
    return value;
}

std::string ConsoleIO::readToken() { return nextLine(true); }

long long ConsoleIO::readNumber(long long min, long long max) {
    return parseNumber(nextLine(true), min, max);
}

std::string ConsoleIO::readLine() { return nextLine(false); }

long long ConsoleIO::readNumberNoCancel(long long min, long long max) {
    return parseNumber(nextLine(false), min, max);
}

// ── 파일 로드 ───────────────────────────────────────────────────────────
void System::promptAndLoad() {
    while (true) {
        std::cout << "Initial condition file name (Enter = " << DEFAULT_FILE << "): ";
        std::string name = io_.readLine();
        if (name.empty()) name = DEFAULT_FILE;

        std::ifstream probe(name);
        if (!probe) {                                                    // 결정 6: 다시 묻기
            std::cout << "[Error] Cannot open file: " << name << " (please try again)\n";
            continue;
        }
        probe.close();
        load(name);

        std::size_t accountCount = 0;
        for (const auto& bank : banks_) accountCount += bank->accounts().size();
        std::cout << "Loaded " << banks_.size() << " bank(s), " << accountCount << " account(s), "
                  << atms_.size() << " ATM(s) from " << name << '\n';
        return;
    }
}

void System::load(const std::string& filename) {
    std::ifstream fin(filename);
    if (!fin) throw FileFormatException("Cannot open file: " + filename);

    // 공백 단위로 읽음 (CRLF의 '\r'도 공백으로 처리됨)
    auto next = [&fin](const std::string& what) {
        std::string token;
        if (!(fin >> token)) throw FileFormatException("Unexpected end of file while reading " + what);
        return token;
    };
    auto nextNumber = [&next](const std::string& what, long long min, long long max) {
        const std::string token = next(what);
        long long value = 0;
        bool ok = true;
        try {
            std::size_t pos = 0;
            value = std::stoll(token, &pos);
            ok = (pos == token.size());
        } catch (const std::exception&) {
            ok = false;
        }
        if (!ok || value < min || value > max)
            throw FileFormatException("Invalid number for " + what + ": " + token);
        return value;
    };

    const long long bankCount = nextNumber("bank count", 1, 1000);
    const long long accountCount = nextNumber("account count", 0, 100000);
    const long long atmCount = nextNumber("ATM count", 1, 1000);

    std::set<std::string> bankNames, cardKeys, accountKeys, serials;

    // 1) 은행 + 은행별 관리자 카드 (결정 3)
    for (long long i = 0; i < bankCount; ++i) {
        const std::string name = next("bank name");
        if (!bankNames.insert(name).second) throw FileFormatException("Duplicate bank name: " + name);
        banks_.push_back(std::make_unique<Bank>(name));
        AdminCard& admin = banks_.back()->issueAdminCard();
        cardKeys.insert(admin.key());
    }

    // 2) 계좌 + 카드 (REQ1.11 순서: 은행, 이름, 계좌번호, 잔액, 카드 번호, 비밀번호)
    for (long long i = 0; i < accountCount; ++i) {
        const std::string bankName = next("account bank name");
        const std::string owner = next("account user name");
        const std::string accountNumber = next("account number");
        const long long balance = nextNumber("account balance", 0, LLONG_MAX / 4);
        const std::string cardNumber = next("card number");
        const std::string password = next("card password");

        Bank* bank = findBank(bankName);
        if (bank == nullptr) throw FileFormatException("Unknown bank for account " + accountNumber + ": " + bankName);
        if (!accountKeys.insert(text::normalizeKey(accountNumber)).second)
            throw FileFormatException("Duplicate account number: " + accountNumber);
        if (!cardKeys.insert(text::normalizeKey(cardNumber)).second)
            throw FileFormatException("Duplicate card number: " + cardNumber);

        Account& account = bank->openAccount(owner, accountNumber, balance);
        bank->issueCard(cardNumber, account, password);
    }

    // 3) ATM (주거래 은행, 일련번호, 종류, 언어, 초기 현금 4개)
    for (long long i = 0; i < atmCount; ++i) {
        const std::string bankName = next("ATM primary bank");
        const std::string serial = next("ATM serial number");
        const std::string type = text::toLower(next("ATM type"));
        const std::string language = text::toLower(next("ATM language"));
        std::array<int, CashBundle::KINDS> counts{};
        for (std::size_t k = 0; k < counts.size(); ++k)
            counts[k] = static_cast<int>(nextNumber("ATM initial cash", 0, INT_MAX / 2));

        Bank* bank = findBank(bankName);
        if (bank == nullptr) throw FileFormatException("Unknown bank for ATM " + serial + ": " + bankName);
        if (!serials.insert(serial).second) throw FileFormatException("Duplicate ATM serial number: " + serial);
        if (type != "single" && type != "multi") throw FileFormatException("Invalid ATM type for " + serial + ": " + type);
        if (language != "unilingual" && language != "bilingual")
            throw FileFormatException("Invalid ATM language for " + serial + ": " + language);

        const bool bilingual = (language == "bilingual");
        if (type == "single") atms_.push_back(std::make_unique<SingleBankATM>(serial, *bank, bilingual));
        else                  atms_.push_back(std::make_unique<MultiBankATM>(serial, *bank, bilingual));

        // REQ1.4: 주거래 은행이 ATM에 현금을 넣음 (파일의 초기 현금 = 은행이 공급한 현금, 결정 15)
        bank->supplyCash(*atms_.back(), CashBundle(counts));
    }
}

void System::printAdminCards(std::ostream& out) const {
    out << "Admin cards (no password, usable at ATMs of the issuing bank):";
    for (const auto& bank : banks_) {
        if (bank->adminCard() != nullptr) out << "  " << bank->adminCard()->number() << " (" << bank->name() << ")";
    }
    out << '\n';
}

// ── 메인 루프 ───────────────────────────────────────────────────────────
void System::run() {
    while (true) {
        try {
            std::cout << "\n===== Select ATM =====\n";
            for (std::size_t i = 0; i < atms_.size(); ++i) {
                const ATM& atm = *atms_[i];
                std::cout << (i + 1) << ") " << atm.serial() << "  " << atm.primaryBank().name() << "  "
                          << atm.typeName() << "  " << (atm.isBilingual() ? "Bilingual" : "Unilingual") << '\n';
            }
            std::cout << "0) Exit program\n(Enter '/' at any prompt to print the snapshot)\nSelect: ";
            const long long choice = io_.readNumberNoCancel(0, static_cast<long long>(atms_.size()));
            if (choice == 0) {
                std::cout << "Program terminated.\n";
                return;
            }
            atms_[static_cast<std::size_t>(choice - 1)]->startSession(*this);
        } catch (const InputClosedException&) {
            throw;                                                       // main에서 정상 종료
        } catch (const ATMException& e) {
            std::cout << describeError(Messages::english(), e) << '\n';
        } catch (const std::exception& e) {                              // 마지막 방어선
            std::cout << "[Error] Unexpected error: " << e.what() << '\n';
        }
    }
}

void System::printSnapshot(std::ostream& out) const {
    out << "\n===== Snapshot =====\n[ATMs]\n";
    for (const auto& atm : atms_) {
        out << "  ATM " << atm->serial() << " (" << atm->primaryBank().name() << ", " << atm->typeName()
            << "): " << atm->cash().toString() << " | Total " << text::formatWon(atm->cash().totalAmount())
            << " KRW\n";
    }
    out << "[Accounts]\n";
    for (const auto& bank : banks_) {
        for (const auto& acc : bank->accounts()) {
            out << "  " << bank->name() << " " << acc->accountNumber() << " (" << acc->ownerName()
                << "): " << text::formatWon(acc->balance()) << " KRW\n";
        }
    }
    out << "====================\n";
}

// ── 검색 ────────────────────────────────────────────────────────────────
Bank* System::findBank(const std::string& name) const {
    const auto it = std::find_if(banks_.begin(), banks_.end(),
                                 [&name](const std::unique_ptr<Bank>& b) { return b->name() == name; });
    return it == banks_.end() ? nullptr : it->get();
}

Card* System::findCard(const std::string& cardNumber) const {
    for (const auto& bank : banks_) {
        if (Card* card = bank->findCard(cardNumber)) return card;
    }
    return nullptr;
}

Account* System::findAccount(const std::string& accountNumber) const {
    for (const auto& bank : banks_) {
        if (Account* acc = bank->findAccount(accountNumber)) return acc;
    }
    return nullptr;
}

void Account::deposit(long long amount) {
    if (amount <= 0) throw InvalidInputException(MsgId::ERR_INVALID_AMOUNT);
    balance_ += amount;

void Account::withdraw(long long amount) {
    if (amount <= 0) throw InvalidInputException(MsgId::ERR_INVALID_AMOUNT);
    if (balance_ < amount) throw InsufficientFundsException(MsgId::ERR_INSUFFICIENT_FUNDS, balance_);
    balance_ -= amount;
}

    history_.push_back(std::move(tx));

#include "bank/Bank.h"
#include "atm/ATM.h"
#include "common/TextUtil.h"

#include <stdexcept>
#include <utility>

Bank::Bank(const std::string& name) : name_(name) {}

Account& Bank::openAccount(const std::string& ownerName, const std::string& accountNumber,
                           long long balance) {
    auto account = std::make_unique<Account>(*this, ownerName, accountNumber, balance);
    Account& ref = *account;
    accounts_.push_back(std::move(account));
    accountIndex_[ref.key()] = &ref;
    return ref;
}

UserCard& Bank::issueCard(const std::string& cardNumber, Account& account, const std::string& password) {
    auto card = std::make_unique<UserCard>(cardNumber, *this, account, password);
    UserCard& ref = *card;
    cards_[ref.key()] = std::move(card);
    return ref;
}

AdminCard& Bank::issueAdminCard() {
    auto card = std::make_unique<AdminCard>(*this);
    AdminCard& ref = *card;
    cards_[ref.key()] = std::move(card);
    adminCard_ = &ref;
    return ref;
}

void Bank::supplyCash(ATM& atm, const CashBundle& cash) const {
    if (!atm.isPrimary(*this))
        throw std::logic_error("Bank " + name_ + " cannot supply cash to ATM " + atm.serial());
    atm.acceptCash(cash);
}

Card* Bank::findCard(const std::string& cardNumber) const {
    const auto it = cards_.find(text::normalizeKey(cardNumber));
    return it == cards_.end() ? nullptr : it->second.get();
}

Account* Bank::findAccount(const std::string& accountNumber) const {
    const auto it = accountIndex_.find(text::normalizeKey(accountNumber));
    return it == accountIndex_.end() ? nullptr : it->second;
}

bool Bank::verifyPassword(const UserCard& card, const std::string& password) const {
    // 이 은행이 발급한 카드만 확인 (REQ3.3: 은행이 검증해서 결과를 돌려줌)
    return &card.issuer() == this && card.password_ == password;

#ifndef CARD_H
#define CARD_H

#include <string>

class Bank;
class Account;
class ATM;
class UserCard;

// 카드 (용어 정의: 관리자 카드는 "특수한 종류의 카드", 결정 5)
// 카드는 정보와 "이 ATM에서 쓸 수 있는가" 판단만 담당. session 진행은 ATM이 함
class Card {
public:
    virtual ~Card() = default;

    const std::string& number() const { return number_; }   // 표시용

// ── session ─────────────────────────────────────────────────────────────
void ATM::startSession(System& system) {
    ConsoleIO& io = system.io();
    const Messages* msg = &Messages::english();          // Unilingual은 영어 고정 (REQ1.3)
    bool cardInserted = false;

    std::cout << "\n===== ATM " << serial_ << " (" << primaryBank_->name() << ", " << typeName()
              << ", " << (bilingual_ ? "Bilingual" : "Unilingual") << ") =====\n";
    try {
        if (bilingual_) {                                 // 결정 12: 카드 투입 전 언어 선택
            while (true) {                                // session 시작 전이므로 잘못 입력하면 다시 묻기 (결정 16)
                std::cout << msg->get(MsgId::SELECT_LANGUAGE) << '\n' << msg->get(MsgId::SELECT_PROMPT);
                try {
                    if (io.readNumber(1, 2) == 2) msg = &Messages::korean();
                    break;
                } catch (const InvalidInputException& e) {
                    std::cout << describeError(*msg, e) << '\n';
                }
            }
        }
        std::cout << msg->get(MsgId::CANCEL_HINT) << '\n' << msg->get(MsgId::INSERT_CARD);
        const std::string cardNumber = io.readToken();
        cardInserted = true;                              // REQ2.1: 카드 투입 = session 시작

        Card* card = system.findCard(cardNumber);
        if (card == nullptr || !card->canBeUsedAt(*this))  // REQ3.1
            throw InvalidCardException(MsgId::ERR_INVALID_CARD);

        if (card->isAdmin()) {
            runAdminSession(system, *msg);
        } else {
            UserCard* user = card->asUserCard();
            if (user == nullptr) throw InvalidCardException(MsgId::ERR_INVALID_CARD);
            if (!authenticate(system, *user, *msg)) throw ATMException(MsgId::ERR_PASSWORD_LOCKED);
            runUserSession(system, *user, *msg);
        }
    } catch (const ATMException& e) {
        std::cout << describeError(*msg, e) << '\n';
    } catch (const CancelException&) {
        std::cout << msg->get(MsgId::CANCELLED) << '\n';
    }
    if (cardInserted) std::cout << msg->get(MsgId::CARD_RETURNED) << '\n';
}

bool ATM::authenticate(System& system, UserCard& card, const Messages& msg) {
    for (int attempt = 1; attempt <= FeePolicy::MAX_PASSWORD_ATTEMPTS; ++attempt) {
        std::cout << msg.get(MsgId::ENTER_PASSWORD);
        const std::string password = system.io().readToken();
        if (card.issuer().verifyPassword(card, password)) return true;     // REQ3.3: 은행이 확인
        const int left = FeePolicy::MAX_PASSWORD_ATTEMPTS - attempt;
        if (left > 0)                                                       // REQ3.4
            std::cout << describeError(msg, ATMException(MsgId::ERR_WRONG_PASSWORD, left)) << '\n';
    }
    return false;                                                           // REQ3.5
}

void ATM::runAdminSession(System& system, const Messages& msg) {
    std::cout << msg.get(MsgId::ADMIN_MENU) << '\n' << msg.get(MsgId::SELECT_PROMPT);   // REQ7.1
    system.io().readNumber(1, 1);

    system.log().printForAtm(std::cout, msg, serial_);                     // REQ7.2: 이 ATM의 거래 (결정 1)
    const std::string path = "ATM_" + serial_ + "_transaction_history.txt";
    if (system.log().saveForAtm(path, msg, serial_))                       // REQ7.3
        std::cout << msg.format(MsgId::HISTORY_SAVED, path) << '\n';
}

}

#include <string>

// 최소 장수 출금 계산 결과
enum class DispenseResult { OK, NOT_POSSIBLE, TOO_MANY_BILLS };

// 권종별 지폐 장수 묶음 (REQ1.10). ATM 보유 현금, 투입 현금, 배출 현금 모두 이 타입
// index 순서: [0]=50,000 [1]=10,000 [2]=5,000 [3]=1,000
class CashBundle {
public:
    static constexpr int KINDS = 4;
    static constexpr std::array<int, KINDS> DENOMINATIONS = { 50000, 10000, 5000, 1000 };

    int totalBills() const;              // 전체 장수
    long long totalAmount() const;       // 전체 금액

    void add(const CashBundle& other);
    void remove(const CashBundle& other);       // 부족하면 InsufficientCashException
    bool contains(const CashBundle& other) const;

    // 최소 장수로 amount를 만드는 조합 계산 (REQ5.1). 이 객체는 바꾸지 않음
    // 큰 권종이 작은 권종의 배수이므로 큰 권종부터 최대한 쓰는 greedy로 최소 장수가 됨
    DispenseResult planDispense(long long amount, int maxBills, CashBundle& out) const;

    // "50,000x2 10,000x3 ..." (skipZero면 0장인 권종 생략)
    std::string toString(bool skipZero = false) const;

    std::array<int, KINDS> counts_{};

#include <numeric>

CashBundle::CashBundle(const std::array<int, KINDS>& counts) : counts_(counts) {}

int CashBundle::count(int index) const { return counts_.at(static_cast<std::size_t>(index)); }

int CashBundle::totalBills() const {
    return std::accumulate(counts_.begin(), counts_.end(), 0);
}

long long CashBundle::totalAmount() const {
    return std::inner_product(counts_.begin(), counts_.end(), DENOMINATIONS.begin(), 0LL,
                              std::plus<>(),
                              [](int count, int denom) { return static_cast<long long>(count) * denom; });
}

void CashBundle::add(const CashBundle& other) {
    for (std::size_t i = 0; i < counts_.size(); ++i) counts_[i] += other.counts_[i];
}

bool CashBundle::contains(const CashBundle& other) const {
    for (std::size_t i = 0; i < counts_.size(); ++i) {
        if (counts_[i] < other.counts_[i]) return false;
    }
    return true;
}

void CashBundle::remove(const CashBundle& other) {
    if (!contains(other)) throw InsufficientCashException(MsgId::ERR_INSUFFICIENT_CASH);
    for (std::size_t i = 0; i < counts_.size(); ++i) counts_[i] -= other.counts_[i];
}

DispenseResult CashBundle::planDispense(long long amount, int maxBills, CashBundle& out) const {
    if (amount <= 0) return DispenseResult::NOT_POSSIBLE;

    std::array<int, KINDS> plan{};
    long long remaining = amount;
    for (std::size_t i = 0; i < counts_.size(); ++i) {
        const long long denom = DENOMINATIONS[i];
        const long long take = std::min<long long>(counts_[i], remaining / denom);
        plan[i] = static_cast<int>(take);
        remaining -= take * denom;
    }
    if (remaining != 0) return DispenseResult::NOT_POSSIBLE;

    CashBundle result(plan);
    if (result.totalBills() > maxBills) return DispenseResult::TOO_MANY_BILLS;
    out = result;
    return DispenseResult::OK;
}

std::string CashBundle::toString(bool skipZero) const {
    std::string out;
    for (std::size_t i = 0; i < counts_.size(); ++i) {
        if (skipZero && counts_[i] == 0) continue;
        if (!out.empty()) out += ' ';
        out += text::formatWon(DENOMINATIONS[i]) + "x" + std::to_string(counts_[i]);
    }
    return out.empty() ? "-" : out;
}

        throw LimitExceededException(MsgId::ERR_TOO_MANY_BILLS, atm.maxBillsPerTransaction());
    }
}

void TransactionLog::printForAtm(std::ostream& out, const Messages& msg, const std::string& atmSerial) const {
    out << msg.format(MsgId::HISTORY_TITLE, atmSerial) << '\n';
    bool any = false;
    for (const auto& tx : entries_) {
        if (tx->atmSerial() != atmSerial) continue;
        out << tx->describe(msg) << '\n';             // 거래 종류별 describe (polymorphism)
        any = true;

}

    printForAtm(file, msg, atmSerial);

// ── session 중 오류 (REQ9.1) ─────────────────────────────────────────────
// 문구 대신 MsgId를 담음 → 받는 쪽이 선택된 언어로 출력 (REQ8.2)
// detail: 문구의 "{}"에 들어갈 숫자 (잔액, 한도 등). 없으면 -1
class ATMException : public std::runtime_error {
public:
    explicit ATMException(MsgId id, long long detail = -1)
        : std::runtime_error("ATMException(id=" + std::to_string(static_cast<int>(id)) + ")"),
          id_(id), detail_(detail) {}

    MsgId id() const { return id_; }
    long long detail() const { return detail_; }

private:
    MsgId id_;
    long long detail_;
};

class InvalidCardException : public ATMException {         // REQ3.1~3.2
public:

};

public:

};

// 선택된 언어로 오류 문구 만들기: "[Error] ..." / "[오류] ..."
inline std::string describeError(const Messages& msg, const ATMException& e) {
    std::string body = e.detail() >= 0 ? msg.format(e.id(), e.detail()) : msg.get(e.id());
    return msg.get(MsgId::ERROR_PREFIX) + body;
}

#include "common/TextUtil.h"

#include <algorithm>
#include <cctype>

namespace text {

std::string trim(const std::string& s) {
    const char* ws = " \t\r\n";
    const std::size_t begin = s.find_first_not_of(ws);
    if (begin == std::string::npos) return "";
    const std::size_t end = s.find_last_not_of(ws);
    return s.substr(begin, end - begin + 1);
}

std::string toLower(const std::string& s) {
    std::string result = s;
    std::transform(result.begin(), result.end(), result.begin(),
                   [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
    return result;
}

std::string normalizeKey(const std::string& s) {
    std::string result;
    for (unsigned char c : s) {
        if (c == '-' || std::isspace(c)) continue;
        result.push_back(static_cast<char>(std::toupper(c)));
    }
    return result;
}

std::string formatWon(long long value) {
    const bool negative = value < 0;
    const unsigned long long magnitude = negative
        ? 0ULL - static_cast<unsigned long long>(value)
        : static_cast<unsigned long long>(value);
    const std::string digits = std::to_string(magnitude);

    std::string out;
    int count = 0;
    for (auto it = digits.rbegin(); it != digits.rend(); ++it) {
        if (count > 0 && count % 3 == 0) out.push_back(',');
        out.push_back(*it);
        ++count;
    }
    if (negative) out.push_back('-');
    std::reverse(out.begin(), out.end());
    return out;
}

}  // namespace text
